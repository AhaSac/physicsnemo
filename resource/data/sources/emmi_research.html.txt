document.addEventListener("DOMContentLoaded", function() {
  // Select all .nav_dropdown_text elements that are NOT inside .nav_dropdown_content_left
  const links = document.querySelectorAll(".nav_dropdown_text:not(.nav_dropdown_content_left .nav_dropdown_text)");

  links.forEach(link => {
    link.addEventListener("mouseenter", () => {
      links.forEach(l => {
        if (l !== link) l.style.color = "#a3a3a3"; // color for other links
      });
    });

    link.addEventListener("mouseleave", () => {
      links.forEach(l => l.style.color = ""); // reset to original color
    });
  });
});
